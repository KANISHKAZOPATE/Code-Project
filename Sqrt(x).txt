class Solution {
    public int mySqrt(int x) {
        if (x < 2) return x; // square root of 0 or 1 is itself

        int left = 1, right = x / 2, result = 0;
        
        while (left <= right) {
            int mid = left + (right - left) / 2;

            // Use long to avoid overflow when multiplying mid * mid
            long square = (long) mid * mid;
            
            if (square == x) {
                return mid; // perfect square
            } else if (square < x) {
                result = mid;      // mid could be the answer, move right
                left = mid + 1;
            } else {
                right = mid - 1;   // mid is too big
            }
        }
        
        return result;
    }
}
