class Solution {
    public void solveSudoku(char[][] board) {
        solve(board);
    }
    
    private boolean solve(char[][] board) {
        // Try to find the next empty spot ('.')
        for (int i = 0; i < 9; i++) {
            for (int j = 0; j < 9; j++) {
                if (board[i][j] == '.') {
                    // Try all numbers from 1 to 9
                    for (char c = '1'; c <= '9'; c++) {
                        // If the current number is valid, place it
                        if (isValid(board, i, j, c)) {
                            board[i][j] = c;
                            
                            // Recursively try to solve the rest of the board
                            if (solve(board)) {
                                return true; // Found a valid solution
                            }
                            
                            // Backtrack if the current choice doesn't work
                            board[i][j] = '.';
                        }
                    }
                    // If no valid number can be placed, return false to backtrack
                    return false;
                }
            }
        }
        return true; // All cells are filled, valid solution
    }
    
    // Check if placing a number at board[row][col] is valid
    private boolean isValid(char[][] board, int row, int col, char c) {
        // Check the row
        for (int i = 0; i < 9; i++) {
            if (board[row][i] == c) {
                return false;
            }
        }
        
        // Check the column
        for (int i = 0; i < 9; i++) {
            if (board[i][col] == c) {
                return false;
            }
        }
        
        // Check the 3x3 sub-grid
        int startRow = (row / 3) * 3;
        int startCol = (col / 3) * 3;
        for (int i = startRow; i < startRow + 3; i++) {
            for (int j = startCol; j < startCol + 3; j++) {
                if (board[i][j] == c) {
                    return false;
                }
            }
        }
        
        return true;
    }
}
