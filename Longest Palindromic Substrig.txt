class Solution {
    public String longestPalindrome(String s) {
        if (s == null || s.length() == 0) {
            return "";
        }

        int start = 0;
        int end = 0;

        // Function to expand the palindrome from the center
        for (int i = 0; i < s.length(); i++) {
            // Check for odd length palindrome (center is at i)
            int len1 = expandAroundCenter(s, i, i);
            // Check for even length palindrome (center is between i and i+1)
            int len2 = expandAroundCenter(s, i, i + 1);

            // Find the maximum length palindrome
            int len = Math.max(len1, len2);

            // If we found a longer palindrome, update start and end
            if (len > end - start) {
                start = i - (len - 1) / 2;
                end = i + len / 2;
            }
        }

        return s.substring(start, end + 1);
    }

    // Helper function to expand the palindrome around the center
    private int expandAroundCenter(String s, int left, int right) {
        while (left >= 0 && right < s.length() && s.charAt(left) == s.charAt(right)) {
            left--;
            right++;
        }
        return right - left - 1;
    }
}
