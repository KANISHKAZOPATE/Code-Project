class Solution {
    public String multiply(String num1, String num2) {
        if (num1.equals("0") || num2.equals("0")) return "0";

        int m = num1.length();
        int n = num2.length();
        int[] result = new int[m + n]; // The result can be at most m+n digits long

        // Multiply each digit from num1 and num2
        for (int i = m - 1; i >= 0; i--) {
            int d1 = num1.charAt(i) - '0';
            for (int j = n - 1; j >= 0; j--) {
                int d2 = num2.charAt(j) - '0';
                int product = d1 * d2;
                int p1 = i + j;     // carry position
                int p2 = i + j + 1; // result position

                int sum = product + result[p2];
                result[p2] = sum % 10;       // store unit digit
                result[p1] += sum / 10;      // add carry to the left
            }
        }

        // Convert result array to string, skipping leading zeros
        StringBuilder sb = new StringBuilder();
        for (int num : result) {
            if (!(sb.length() == 0 && num == 0)) { // skip leading zeros
                sb.append(num);
            }
        }

        return sb.toString();
    }
}
