class Solution {
    public int romanToInt(String s) {
        // Mapping of Roman symbols to their integer values
        Map<Character, Integer> romanMap = new HashMap<>();
        romanMap.put('I', 1);
        romanMap.put('V', 5);
        romanMap.put('X', 10);
        romanMap.put('L', 50);
        romanMap.put('C', 100);
        romanMap.put('D', 500);
        romanMap.put('M', 1000);

        int total = 0;
        int prevValue = 0;

        // Iterate from end of the string to start
        for (int i = s.length() - 1; i >= 0; i--) {
            int currentValue = romanMap.get(s.charAt(i));

            if (currentValue < prevValue) {
                // Subtract if smaller value comes before a larger one
                total -= currentValue;
            } else {
                // Otherwise, add the value
                total += currentValue;
            }

            prevValue = currentValue;
        }

        return total;
    }
}
